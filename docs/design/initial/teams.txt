<script>
        document.querySelectorAll('input').forEach(input => {
            input.addEventListener('change', (e) => {
                const value = e.target.value;
                const container = e.target.closest('.flex.items-center.gap-12');
                const label = container.querySelector('.text-cream-gold');
                const bar = container.querySelector('.progress-bar-fill');
                
                label.innerText = `${value}% Share`;
                const percentage = Math.min(Math.max(value, 0), 100);
                bar.style.width = `${percentage}%`;
                
                // Spring effect on update
                bar.style.transition = 'none';
                bar.style.transform = 'scaleX(1.1)';
                setTimeout(() => {
                    bar.style.transition = 'all 0.6s cubic-bezier(0.175, 0.885, 0.32, 1.275)';
                    bar.style.transform = 'scaleX(1)';
                }, 50);
            });
        });

        // Simple button logic for increment/decrement
        document.querySelectorAll('button').forEach(btn => {
            if (btn.innerText === '+' || btn.innerText === '-') {
                btn.addEventListener('click', () => {
                    const input = btn.parentNode.querySelector('input');
                    let val = parseInt(input.value);
                    if (btn.innerText === '+') val += 5;
                    else val -= 5;
                    input.value = val;
                    input.dispatchEvent(new Event('change'));
                });
            }
        });
    </script>
